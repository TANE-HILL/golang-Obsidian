Работа над проектом по итогу Модуля 5.

```Go
package main
import (
    "fmt"
    "errors"
    "sync"
    )

var ErrOrderNotFound = errors.New("order not found")

//~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

type Order struct {
    ID int
    Customer string
    Address string
    IsDelivered bool
}
func (o *Order) MarkDelivered() {
    o.IsDelivered = true
}

//~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

type OrderRepository interface{
    Add(order Order)
    GetByID(id int) (Order, error)
    Update(order Order)error
}

type InMemoryOrderRepo struct {
    Orders map[int]Order
    mu sync.Mutex
}

func (repo *InMemoryOrderRepo) Add(order Order) {
    defer repo.mu.Unlock()
    repo.mu.Lock()
    repo.Orders[order.ID] = order
}

func (repo *InMemoryOrderRepo) GetByID(id int) (Order, error){
    defer repo.mu.Unlock()
    repo.mu.Lock()
    value, ok := repo.Orders[id]
    if !ok {
        return value, ErrOrderNotFound
    }
    return value, nil
}

func (repo *InMemoryOrderRepo) Update(order Order)error{
    defer repo.mu.Unlock()
    repo.mu.Lock()
    if _, ok := repo.Orders[order.ID]; !ok {
        return ErrOrderNotFound
    }
    repo.Orders[order.ID] = order
    return nil
}

//~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

func DeliverMany(repo OrderRepository, ids []int) {
    var wg sync.WaitGroup
    for _, id := range ids {
        wg.Add(1)
        go func(){
            defer wg.Done()
            order, errGet := repo.GetByID(id)
            if errGet != nil {
                fmt.Printf("Ошибка: %s (ID: %v)\n", errGet, id)
                return
            }
            order.MarkDelivered()
            if errUpdate := repo.Update(order); errUpdate != nil {
                fmt.Printf("Ошибка: %s (ID: %v)\n", errUpdate, id)
                return
            }
            fmt.Printf("Updated order %v\n", order.ID)
        }()
    }
    wg.Wait()
}

//~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

func PrintAllOrders(repo *InMemoryOrderRepo){
    for _, v := range repo.Orders {
        fmt.Printf("ID: %v, %v, Delivered: %v\n", v.ID, v.Customer, v.IsDelivered)
    }
}
func NewInMemoryOrderRepo() *InMemoryOrderRepo{
    return &InMemoryOrderRepo{
        Orders: make(map[int]Order),
    }
}

//~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

func main() {
    repo := NewInMemoryOrderRepo()
    order1 := Order{
        ID: 1,
        Customer: "Андрей",
        Address: "40 лет победы, 31-Б",
        IsDelivered: false,
    }
    order2 := Order{
        ID: 2,
        Customer: "Иван",
        Address: "ул. Кирова, 136",
        IsDelivered: false,
    }
    order3 := Order{
        ID: 3,
        Customer: "Анастасия",
        Address: "ул. Васенко, 15",
        IsDelivered: false,
    }
    order4 := Order{
        ID: 4,
        Customer: "Валерий",
        Address: "пос. Западный, д. 9",
        IsDelivered: false,
    }
    order5 := Order{
        ID: 5,
        Customer: "Альберт",
        Address: "ул. Петра Столыпина, д. 112",
        IsDelivered: false,
    }
    order6 := Order{
        ID: 6,
        Customer: "Владислав",
        Address: "Винегретовая ул., д. 9",
        IsDelivered: false,
    }
    order7 := Order{
        ID: 7,
        Customer: "Даниил",
        Address: "ул. Карла Маркса, 54",
        IsDelivered: false,
    }
    repo.Add(order1)
    repo.Add(order2)
    repo.Add(order3)
    repo.Add(order4)
    repo.Add(order5)
    repo.Add(order6)
    repo.Add(order7)
    fmt.Println("[Before]")
    PrintAllOrders(repo)

    fmt.Println("\n[Delivery Result]")
    ids := []int{1, 2, 5, 8}
    DeliverMany(repo, ids)
    
    fmt.Println("\n[After]")
    PrintAllOrders(repo)
}
```
