```Go
package main
import "fmt"
type OrderRepository interface{
    Add(order Order)
    GetByID(id int) (Order, bool)
    Update(order Order)
}
type InMemoryOrderRepo struct {
    Orders map[int]Order
}
type Order struct {
    ID int
    Customer string
    Address string
    IsDelivered bool
}
func (o *Order) MarkDelivered() {
    o.IsDelivered = true
}
func (repo *InMemoryOrderRepo) Add(order Order) {
    repo.Orders[order.ID] = order
}
func (repo *InMemoryOrderRepo) GetByID(id int) (Order, bool){
    value, ok := repo.Orders[id]
    return value, ok
}
func (repo *InMemoryOrderRepo) Update(order Order){
    repo.Orders[order.ID] = order
}
func PrintAllOrders(repo *InMemoryOrderRepo){
    for _, v := range repo.Orders {
        fmt.Printf("ID: %v, %v, Delivered: %v\n", v.ID, v.Customer, v.IsDelivered)
    }
}
func NewInMemoryOrderRepo() *InMemoryOrderRepo{
    return &InMemoryOrderRepo{
        Orders: make(map[int]Order),
    }
}
func main() {
    repo := NewInMemoryOrderRepo()
    order1 := Order{
        ID: 1,
        Customer: "Андрей",
        Address: "40 лет победы, 31-Б",
        IsDelivered: false,
    }
    order2 := Order{
        ID: 2,
        Customer: "Иван",
        Address: "ул. Кирова, 136",
        IsDelivered: false,
    }
    order3 := Order{
        ID: 3,
        Customer: "Анастасия",
        Address: "ул. Васенко, 15",
        IsDelivered: false,
    }
    repo.Add(order1)
    repo.Add(order2)
    repo.Add(order3)
    fmt.Println("[Before]")
    PrintAllOrders(repo)
    order, ok := repo.GetByID(2)
    if ok {
        order.MarkDelivered()
        repo.Update(order)
    }
    fmt.Println("[After]")
    PrintAllOrders(repo)
}
```
