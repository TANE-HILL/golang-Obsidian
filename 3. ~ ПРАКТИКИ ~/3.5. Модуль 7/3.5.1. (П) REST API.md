# <u>Задание</u>

Создать HTTP-сервер на Go, который реализует CRUD-операции (_Create, Read, Update, Delete_) над сущностью **Book**.

#### Модель данных

Каждая книга должна содержать следующие поля:

|Поле|Тип|Описание|
|---|---|---|
|`ID`|`int`|Уникальный идентификатор книги|
|`Title`|`string`|Название книги|
|`Author`|`string`|Имя автора|
|`Available`|`bool`|Признак: доступна ли книга для выдачи|
Все поля должны корректно сериализоваться в JSON (через `json`-теги).

#### REST-эндпоинты

|Метод|Путь|Описание|Успешный ответ|
|---|---|---|---|
|`GET`|`/books`|Получить список всех книг|`200 OK` + массив JSON|
|`GET`|`/books/{id}`|Получить книгу по её ID|`200 OK` + JSON книги|
|`POST`|`/books`|Добавить новую книгу|`201 Created` + JSON созданной книги|
|`PUT`|`/books/{id}`|Обновить данные книги|`200 OK` + обновлённая книга|
|`DELETE`|`/books/{id}`|Удалить книгу по ID|`204 No Content`|
#### Требования к реализации

1. **Сервер:** использовать стандартную библиотеку Go — пакет `net/http`.    
2. **Формат данных:** все запросы и ответы — в формате JSON.
3. **Хранилище:** использовать срез `[]Book` как временную “базу данных”.
4. **Маршрутизация:** можно использовать `http.HandleFunc` и `switch r.Method`.
5. **Проверка метода:** при неправильном HTTP-методе возвращать `405 Method Not Allowed`.
6. **Ошибки:** возвращать читаемые ошибки с подходящим статус-кодом:
    - `400 Bad Request` — неверный ID или битый JSON;
    - `404 Not Found` — книга не найдена;
    - `500 Internal Server Error` — внутренняя ошибка (при необходимости).
7. **Заголовки:** для всех успешных ответов установить
    `Content-Type: application/json`.

# <u>Решение</u>

`books.go`:
```Go
package handlers  
  
import (  
    "encoding/json"  
    "net/http"    "restapi/models"    "strconv"    "strings")  
  
var books = []models.Book{  
    {ID: 1, Title: "Мастер и Маргарита", Author: "Булгаков", Available: true},  
    {ID: 2, Title: "Гарри Поттер", Author: "Роулинг", Available: false},  
}  
  
func GetBooks(w http.ResponseWriter, r *http.Request) {  
    w.Header().Set("Content-Type", "application/json")  
    json.NewEncoder(w).Encode(books)  
}  
  
func GetBookByID(w http.ResponseWriter, r *http.Request) {  
    idStr := strings.TrimPrefix(r.URL.Path, "/books/")  
    id, err := strconv.Atoi(idStr)  
    if err != nil {  
       http.Error(w, "Некорректный ID", http.StatusBadRequest)  
       return  
    }  
  
    for _, t := range books {  
       if t.ID == id {  
          w.Header().Set("Content-Type", "application/json")  
          json.NewEncoder(w).Encode(t)  
          return  
       }  
    }  
    http.Error(w, "Задача не найдена", http.StatusNotFound)  
}  
  
func CreateBook(w http.ResponseWriter, r *http.Request) {  
    var newTask models.Book  
    if err := json.NewDecoder(r.Body).Decode(&newTask); err != nil {  
       http.Error(w, "Невалидный JSON", http.StatusBadRequest)  
       return  
    }  
  
    newTask.ID = len(books) + 1  
    books = append(books, newTask)  
  
    w.Header().Set("Content-Type", "application/json")  
    w.WriteHeader(http.StatusCreated)  
    json.NewEncoder(w).Encode(newTask)  
}  
  
func UpdateBook(w http.ResponseWriter, r *http.Request) {  
    idStr := strings.TrimPrefix(r.URL.Path, "/books/")  
    id, err := strconv.Atoi(idStr)  
    if err != nil {  
       http.Error(w, "Некорректный ID", http.StatusBadRequest)  
       return  
    }  
  
    var updated models.Book  
    if err := json.NewDecoder(r.Body).Decode(&updated); err != nil {  
       http.Error(w, "Невалидный JSON", http.StatusBadRequest)  
       return  
    }  
  
    for i, t := range books {  
       if t.ID == id {  
          books[i].Title = updated.Title  
          books[i].Author = updated.Author  
          books[i].Available = updated.Available  
          w.Header().Set("Content-Type", "application/json")  
          json.NewEncoder(w).Encode(books[i])  
          return  
       }  
    }  
  
    http.Error(w, "Задача не найдена", http.StatusNotFound)  
}  
  
func DeleteBook(w http.ResponseWriter, r *http.Request) {  
    idStr := strings.TrimPrefix(r.URL.Path, "/books/")  
    id, err := strconv.Atoi(idStr)  
    if err != nil {  
       http.Error(w, "Некорректный ID", http.StatusBadRequest)  
       return  
    }  
  
    for i, t := range books {  
       if t.ID == id {  
          books = append(books[:i], books[i+1:]...)  
          w.WriteHeader(http.StatusNoContent)  
          return  
       }  
    }  
  
    http.Error(w, "Задача не найдена", http.StatusNotFound)  
}
```

`book.go`:
```Go
package models  
  
type Book struct {  
    ID        int    `json:"id"`  
    Title     string `json:"title"`  
    Author    string `json:"author"`  
    Available bool   `json:"available"`  
}
```

`main.go`:
```Go
package main  
  
import (  
    "fmt"  
    "net/http"    "restapi/handlers")  
  
func main() {  
    http.HandleFunc("/books", func(w http.ResponseWriter, r *http.Request) {  
       switch r.Method {  
       case http.MethodGet:  
          handlers.GetBooks(w, r)  
       case http.MethodPost:  
          handlers.CreateBook(w, r)  
       default:  
          http.Error(w, "Метод не поддерживается", http.StatusMethodNotAllowed)  
       }  
    })  
  
    http.HandleFunc("/books/", func(w http.ResponseWriter, r *http.Request) {  
       switch r.Method {  
       case http.MethodGet:  
          handlers.GetBookByID(w, r)  
       case http.MethodPut:  
          handlers.UpdateBook(w, r)  
       case http.MethodDelete:  
          handlers.DeleteBook(w, r)  
       default:  
          http.Error(w, "Метод не поддерживается", http.StatusMethodNotAllowed)  
       }  
    })  
  
    fmt.Println("Сервер запущен на http://localhost:8080")  
  
    err := http.ListenAndServe(":8080", nil)  
    if err != nil {  
       fmt.Println("Ошибка запуска сервера:", err)  
    }  
}
```
