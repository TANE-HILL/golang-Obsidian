# <u>Задание</u>

#### 1. Добавить `LoggingMiddleware`

**`middleware/logging.go` должен**:
- принимать `http.Handler` и возвращать `http.Handler`;
- выводить в консоль метод, путь и время обработки запроса;
- передавать запрос дальше в цепочку (`next.ServeHTTP(w, r)`).
- **Подсказка:**
    - используйте `time.Now()` для засечки времени начала;
    - а `time.Since(start)` — для вычисления длительности;
    - пишите лог через `log.Printf()`.

#### 2. Добавить `RecoveryMiddleware`

**`middleware/recovery.go` должен:**
- перехватывать возможные паники внутри обработчиков;
- писать сообщение об ошибке в лог;
- возвращать пользователю ответ с кодом `500` и текстом “Внутренняя ошибка сервера”;
- при этом сервер **не должен падать**.

#### 3. Подключить middleware в `main.go`

# <u>Решение</u>

`main.go`:
```Go
package main  
  
import (  
    "fmt"  
    "net/http"
    "restapi/handlers"
    "restapi/middleware"
)  
  
func main() {  
    mux := http.NewServeMux()  
    mux.HandleFunc("/books", func(w http.ResponseWriter, r *http.Request) {  
       switch r.Method {  
       case http.MethodGet:  
          handlers.GetBooks(w, r)  
       case http.MethodPost:  
          handlers.CreateBook(w, r)  
       default:  
          http.Error(w, "Метод не поддерживается", http.StatusMethodNotAllowed)  
       }  
    })  
  
    mux.HandleFunc("/books/", func(w http.ResponseWriter, r *http.Request) {  
       switch r.Method {  
       case http.MethodGet:  
          handlers.GetBookByID(w, r)  
       case http.MethodPut:  
          handlers.UpdateBook(w, r)  
       case http.MethodDelete:  
          handlers.DeleteBook(w, r)  
       default:  
          http.Error(w, "Метод не поддерживается", http.StatusMethodNotAllowed)  
       }  
    })  
    var handler http.Handler = mux  
    handler = middleware.LoggingMiddleware(handler)  
    handler = middleware.RecoveryMiddleware(handler)  
    fmt.Println("Сервер запущен на http://localhost:8080")  
  
    err := http.ListenAndServe(":8080", handler)  
    if err != nil {  
       fmt.Println("Ошибка запуска сервера:", err)  
    }  
}
```

`books.go`:
```Go
package handlers  
  
import (  
    "encoding/json"  
    "net/http"
    "restapi/models"
    "strconv"
    "strings"
)  
  
var books = []models.Book{  
    {ID: 1, Title: "Мастер и Маргарита", Author: "Булгаков", Available: true},  
    {ID: 2, Title: "Гарри Поттер", Author: "Роулинг", Available: false},  
}  
  
func GetBooks(w http.ResponseWriter, r *http.Request) {  
    w.Header().Set("Content-Type", "application/json")  
    json.NewEncoder(w).Encode(books)  
}  
  
func GetBookByID(w http.ResponseWriter, r *http.Request) {  
    idStr := strings.TrimPrefix(r.URL.Path, "/books/")  
    id, err := strconv.Atoi(idStr)  
    if err != nil {  
       http.Error(w, "Некорректный ID", http.StatusBadRequest)  
       return  
    }  
  
    for _, t := range books {  
       if t.ID == id {  
          w.Header().Set("Content-Type", "application/json")  
          json.NewEncoder(w).Encode(t)  
          return  
       }  
    }  
    http.Error(w, "Книга не найдена", http.StatusNotFound)  
}  
  
func CreateBook(w http.ResponseWriter, r *http.Request) {  
    var newBook models.Book  
    if err := json.NewDecoder(r.Body).Decode(&newBook); err != nil {  
       http.Error(w, "Невалидный JSON", http.StatusBadRequest)  
       return  
    }  
  
    newBook.ID = len(books) + 1  
    books = append(books, newBook)  
  
    w.Header().Set("Content-Type", "application/json")  
    w.WriteHeader(http.StatusCreated)  
    json.NewEncoder(w).Encode(newBook)  
}  
  
func UpdateBook(w http.ResponseWriter, r *http.Request) {  
    idStr := strings.TrimPrefix(r.URL.Path, "/books/")  
    id, err := strconv.Atoi(idStr)  
    if err != nil {  
       http.Error(w, "Некорректный ID", http.StatusBadRequest)  
       return  
    }  
  
    var updated models.Book  
    if err := json.NewDecoder(r.Body).Decode(&updated); err != nil {  
       http.Error(w, "Невалидный JSON", http.StatusBadRequest)  
       return  
    }  
  
    for i, t := range books {  
       if t.ID == id {  
          books[i].Title = updated.Title  
          books[i].Author = updated.Author  
          books[i].Available = updated.Available  
          w.Header().Set("Content-Type", "application/json")  
          json.NewEncoder(w).Encode(books[i])  
          return  
       }  
    }  
  
    http.Error(w, "Книга не найдена", http.StatusNotFound)  
}  
  
func DeleteBook(w http.ResponseWriter, r *http.Request) {  
    idStr := strings.TrimPrefix(r.URL.Path, "/books/")  
    id, err := strconv.Atoi(idStr)  
    if err != nil {  
       http.Error(w, "Некорректный ID", http.StatusBadRequest)  
       return  
    }  
  
    for i, t := range books {  
       if t.ID == id {  
          books = append(books[:i], books[i+1:]...)  
          w.WriteHeader(http.StatusNoContent)  
          return  
       }  
    }  
  
    http.Error(w, "Книга не найдена", http.StatusNotFound)  
}
```

#### Middleware:

`logging.go`:
```Go
package middleware  
  
import (  
    "log"  
    "net/http"
    "time"
)  
  
func LoggingMiddleware(next http.Handler) http.Handler {  
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {  
       start := time.Now()  
       next.ServeHTTP(w, r)  
       log.Printf("%s %s - %v", r.Method, r.URL.Path, time.Since(start))  
    })  
}
```

`recovery.go`:
```Go
package middleware  
  
import (  
    "log"  
    "net/http")  
  
func RecoveryMiddleware(next http.Handler) http.Handler {  
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {  
       defer func() {  
          if err := recover(); err != nil {  
             log.Printf("Паника: %v", err)  
             http.Error(w, "Внутренняя ошибка сервера", http.StatusInternalServerError)  
          }  
       }()  
       next.ServeHTTP(w, r)  
    })  
}
```
