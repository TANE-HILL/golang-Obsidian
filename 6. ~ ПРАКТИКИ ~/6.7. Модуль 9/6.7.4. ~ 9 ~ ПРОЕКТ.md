- Структура таблицы была описана только в SQL, но в коде не использовалась;
- Проект не был подключён к реальной СУБД.

Теперь:

- Мы создаём хранилище `PostgresOrderRepo`, которое работает через базу данных;
- Подключаемся к PostgreSQL с использованием настроек из `.env`;
- Заменяем in-memory хранилище на PostgreSQL;
- Реализуем стандартные методы: `Add`, `GetByID`, `GetAll`, `Update`.

## `.../repository/postgres.go`:
```Go
package repository  
  
import (  
    "database/sql"  
    "fmt"
    "gotracker/internal/order
    )  
  
type PostgresOrderRepo struct {  
    db *sql.DB  
}  
  
func NewPostgresOrderRepo(db *sql.DB) *PostgresOrderRepo {  
    return &PostgresOrderRepo{db: db}  
}  
  
func (r *PostgresOrderRepo) Add(order order.Order) error {  
    query := `INSERT INTO orders (id, customer, address, is_delivered) VALUES ($1, $2, $3, $4)`  
    _, err := r.db.Exec(  
       query,  
       order.ID,  
       order.Customer,  
       order.Address,  
       order.IsDelivered,  
    )  
    return err  
}  
  
func (r *PostgresOrderRepo) GetByID(id int) (order.Order, error) {  
    var o order.Order  
    query := `SELECT id, customer, address, is_delivered FROM orders WHERE id = $1`  
    err := r.db.QueryRow(query, id).Scan(&o.ID, &o.Customer, &o.Address, &o.IsDelivered)  
    if err == sql.ErrNoRows {  
       return o, order.ErrOrderNotFound  
    }  
    return o, err  
}  
  
func (r *PostgresOrderRepo) GetAll() []order.Order {  
    var orders []order.Order  
    query := `SELECT id, customer, address, is_delivered FROM orders`  
    rows, err := r.db.Query(query)  
    if err != nil {  
       fmt.Println("Ошибка при получении заказов:", err)  
       return nil  
    }  
    defer rows.Close()  
    for rows.Next() {  
       var o order.Order  
       err := rows.Scan(  
          &o.ID,  
          &o.Customer,  
          &o.Address,  
          &o.IsDelivered,  
       )  
       if err != nil {  
          fmt.Println("Ошибка при чтении заказа:", err)  
          continue  
       }  
       orders = append(orders, o)  
    }  
    if err := rows.Err(); err != nil {  
       fmt.Println("Ошибка при чтении результатов:", err)  
       return nil  
    }  
    return orders  
}  
  
func (r *PostgresOrderRepo) Update(o order.Order) error {  
    query := `UPDATE orders SET customer = $1, address = $2, is_delivered = $3 WHERE id = $4`  
    result, err := r.db.Exec(  
       query,  
       o.Customer,  
       o.Address,  
       o.IsDelivered,  
       o.ID,  
    )  
    if err != nil {  
       return err  
    }  
    count, err := result.RowsAffected()  
    if err != nil {  
       return err  
    }  
    if count == 0 {  
       return order.ErrOrderNotFound  
    }  
    return err  
}
```

## Обновленный `main.go`:
```Go
package main  
  
import (  
    "database/sql"  
    "fmt"
    apphttp "gotracker/internal/http"
    "gotracker/internal/repository"
    "gotracker/internal/service"
    "log"
    "net/http"
    "os"
    
    "github.com/joho/godotenv"
    _ "github.com/lib/pq"  
)  
  
func main() {  
    if err := godotenv.Load(); err != nil {  
       log.Fatal("Ошибка загрузки .env:", err)  
    }  
    dsn := fmt.Sprintf("host=%s port=%s user=%s password=%s dbname=%s sslmode=disable",  
       os.Getenv("DB_HOST"),  
       os.Getenv("DB_PORT"),  
       os.Getenv("DB_USER"),  
       os.Getenv("DB_PASSWORD"),  
       os.Getenv("DB_NAME"),  
    )  
    db, err := sql.Open("postgres", dsn)  
    if err != nil {  
       log.Fatal(err)  
    }  
    if err := db.Ping(); err != nil {  
       log.Fatal("Не удалось подключиться к БД:", err)  
    }  
  
    repo := repository.NewPostgresOrderRepo(db)  
    svc := service.NewOrderService(repo)  
    handler := apphttp.NewHandler(svc)  
  
    log.Println("Сервер запущен на http://localhost:8080")  
    err = http.ListenAndServe(":8080", handler)  
    if err != nil {  
       log.Fatal(err)  
    }  
}
```
