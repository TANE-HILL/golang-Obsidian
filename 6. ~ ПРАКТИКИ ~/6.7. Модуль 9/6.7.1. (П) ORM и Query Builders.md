# Что нужно реализовать

### 1. Таблица `transactions`

Файл: `migrations/002_create_transactions.sql`
Таблица для хранения финансовых операций:
```sql
CREATE TABLE IF NOT EXISTS transactions (
    id SERIAL PRIMARY KEY,
    category_id INT REFERENCES categories(id) ON DELETE CASCADE,
    amount NUMERIC(10,2) NOT NULL,
    note TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);
```
- `category_id` — ссылка на категорию (`income` / `expense`);
- `amount` — сумма (можно положительную или отрицательную);
- `note` — описание, например “обед в кафе”;
- `created_at` — автоматически ставится при добавлении записи.

---
### 2. Модель `Transaction`

Файл: `internal/models/transaction.go`
Структура, которая будет описывать транзакцию:
```go
package models

type Transaction struct {
    ID         int     `db:"id" json:"id"`
    CategoryID int     `db:"category_id" json:"category_id"`
    Amount     float64 `db:"amount" json:"amount"`
    Note       string  `db:"note" json:"note"`
    CreatedAt  string  `db:"created_at" json:"created_at"`
}

```

---
### 3. Репозиторий `TransactionRepo`

Файл: `internal/db/transaction_repo.go`

Добавь методы для добавления и получения транзакций.

Требования:
- Метод `Create()` должен вставлять новую запись в БД и возвращать результат.
- Метод `List()` — выбирать все транзакции, отсортированные по дате.

Подсказка: используй `r.db.GetContext()` и `r.db.SelectContext()` — они делают то же самое, что и `QueryRow`/`Query`, но сразу маппят данные в структуру.

---
### 4. Хендлеры `/transactions`

Файл: `internal/handlers/transactions.go`

Реализуй два хендлера:
1. `POST /transactions` — добавляет транзакцию
	(JSON с `category_id`, `amount`, `note`).
2. `GET /transactions` — возвращает список всех транзакций.

---
### 5. Регистрация маршрута

Файл: `cmd/api/main.go`

Новый блок для `/transactions`:
```go
txs := handlers.NewTransactionHandlers(dbase)

http.HandleFunc("/transactions", func(w http.ResponseWriter, r *http.Request) {
    switch r.Method {
    case http.MethodGet:
        txs.ListTransactions(w, r)
    case http.MethodPost:
        txs.CreateTransaction(w, r)
    default:
        http.Error(w, "Method Not Allowed", http.StatusMethodNotAllowed)
    }
})
```

## `/handlers/transactions.go`:
```Go
package handlers  
  
import (  
    "encoding/json"  
    "net/http"    "strings"  
    "gofinance/internal/db"  
    "github.com/jmoiron/sqlx"
)  
  
type TransactionHandlers struct {  
    repo *db.TransactionRepo  
}  
  
func NewTransactionHandlers(dbase *sqlx.DB) *TransactionHandlers {  
    return &TransactionHandlers{repo: db.NewTransactionRepo(dbase)}  
}  
  
func (h *TransactionHandlers) CreateTransaction(w http.ResponseWriter, r *http.Request) {  
    if r.Method != http.MethodPost {  
       http.Error(w, "Method Not Allowed", http.StatusMethodNotAllowed)  
       return  
    }  
    var in struct {  
       CategoryID int     `json:"category_id"`  
       Amount     float64 `json:"amount"`  
       Note       string  `json:"note"`  
    }  
  
    if err := json.NewDecoder(r.Body).Decode(&in); err != nil {  
       http.Error(w, "Invalid JSON", http.StatusBadRequest)  
       return  
    }  
    in.Note = strings.TrimSpace(in.Note)  
    cat, err := h.repo.Create(r.Context(), in.CategoryID, in.Amount, in.Note)  
    if err != nil {  
       http.Error(w, "Database error: "+err.Error(), http.StatusInternalServerError)  
       return  
    }  
    w.Header().Set("Content-Type", "application/json")  
    w.WriteHeader(http.StatusCreated)  
    json.NewEncoder(w).Encode(cat)  
}  
func (h *TransactionHandlers) ListTransactions(w http.ResponseWriter, r *http.Request) {  
    if r.Method != http.MethodGet {  
       http.Error(w, "Method Not Allowed", http.StatusMethodNotAllowed)  
       return  
    }  
    transactions, err := h.repo.List(r.Context())  
    if err != nil {  
       http.Error(w, "Database error: "+err.Error(), http.StatusInternalServerError)  
       return  
    }  
    w.Header().Set("Content-Type", "application/json")  
    json.NewEncoder(w).Encode(transactions)  
}
```

## `db/transaction_repo.go`:
```Go
package db
  
import (  
    "context"  
    "gofinance/internal/models"  
    "github.com/jmoiron/sqlx"
)  
  
type TransactionRepo struct {  
    db *sqlx.DB  
}  
  
func NewTransactionRepo(db *sqlx.DB) *TransactionRepo {  
    return &TransactionRepo{db: db}  
}  
  
func (r *TransactionRepo) Create(  
    ctx context.Context,  
    categoryID int,  
    amount float64,  
    note string,  
) (models.Transaction, error) {  
    var transaction models.Transaction  
    query := `  
    INSERT INTO transactions (category_id, amount, note)    VALUES ($1, $2, $3)    RETURNING id, category_id, amount, note, created_at;    `  
    if err := r.db.GetContext(  
       ctx,  
       &transaction,  
       query,  
       categoryID,  
       amount,  
       note,  
    ); err != nil {  
       return models.Transaction{}, err  
    }  
    return transaction, nil  
}  
  
func (r *TransactionRepo) List(ctx context.Context) ([]models.Transaction, error) {  
    var transactions []models.Transaction  
  
    err := r.db.SelectContext(ctx, &transactions, `  
       SELECT id, category_id, amount, note, created_at       FROM transactions       ORDER BY created_at;    `)  
    return transactions, err  
}
```
