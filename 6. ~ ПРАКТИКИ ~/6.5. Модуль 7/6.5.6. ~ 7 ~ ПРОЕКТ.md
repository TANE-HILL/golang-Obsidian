```Go
package main  
  
import (  
    apphttp "gotracker/internal/http"  
    "gotracker/internal/repository"
    "gotracker/internal/service"
    "log"
    "net/http"
    )  
  
func main() {  
    repo := repository.NewInMemoryOrderRepo()  
    svc := service.NewOrderService(repo)  
    handler := apphttp.NewHandler(svc)  
  
    log.Println("Сервер запущен на http://localhost:8080")  
    err := http.ListenAndServe(":8080", handler)  
    if err != nil {  
       log.Fatal(err)  
    }  
}
```

# **`server.go`**:
```Go
package http  
  
import (  
    "encoding/json"  
    "gotracker/internal/order"
    "gotracker/internal/service"
    "net/http"
    "strconv"
    "strings"
    )
  
type Handler struct {  
    service *service.OrderService  
}  
  
func NewHandler(service *service.OrderService) *Handler {  
    return &Handler{service: service}  
}  
  
func (h *Handler) ServeHTTP(w http.ResponseWriter, r *http.Request) {  
    if r.URL.Path == "/" && r.Method == http.MethodGet {  
       w.Write([]byte("Добро пожаловать в GoTracker API"))  
       return  
    }  
    path := r.URL.Path  
    method := r.Method  
  
    if path == "/orders" && method == http.MethodGet {  
       h.handleGetAll(w, r)  
       return  
    }  
  
    if path == "/orders" && method == http.MethodPost {  
       h.handleCreate(w, r)  
       return  
    }  
  
    if strings.HasPrefix(path, "/orders/") && method == http.MethodPut {  
       h.handleUpdate(w, r)  
       return  
    }  
  
    http.NotFound(w, r)  
}  
  
func (h *Handler) handleGetAll(w http.ResponseWriter, r *http.Request) {  
    orders := h.service.GetAll()  
    w.Header().Set("Content-Type", "application/json")  
    json.NewEncoder(w).Encode(orders)  
}  
  
func (h *Handler) handleCreate(w http.ResponseWriter, r *http.Request) {  
    var o order.Order  
    if err := json.NewDecoder(r.Body).Decode(&o); err != nil {  
       http.Error(w, "Invalid JSON", http.StatusBadRequest)  
       return  
    }  
    if o.ID <= 0 || o.Customer == "" || o.Address == "" {  
       http.Error(w, "Missing or invalid fields", http.StatusBadRequest)  
       return  
    }  
    h.service.AddOrder(o)  
    w.WriteHeader(http.StatusCreated)  
}  
  
func (h *Handler) handleUpdate(w http.ResponseWriter, r *http.Request) {  
    parts := strings.Split(r.URL.Path, "/")  
    if len(parts) != 3 {  
       http.Error(w, "Invalid URL", http.StatusBadRequest)  
       return  
    }  
    id, err := strconv.Atoi(parts[2])  
    if err != nil || id <= 0 {  
       http.Error(w, "Invalid ID", http.StatusBadRequest)  
       return  
    }  
  
    var o order.Order  
    if err := json.NewDecoder(r.Body).Decode(&o); err != nil {  
       http.Error(w, "Invalid JSON", http.StatusBadRequest)  
       return  
    }  
    o.ID = id  
  
    if err := h.service.UpdateOrder(o); err != nil {  
       http.Error(w, err.Error(), http.StatusNotFound)  
       return  
    }  
    w.WriteHeader(http.StatusOK)  
}
```

# **`inmemory.go`**:
```Go
package repository  
  
import (  
    "gotracker/internal/order"  
    "sync"
    )  
  
type InMemoryOrderRepo struct {  
    Orders map[int]order.Order  
    mu     sync.Mutex  
}  
  
func NewInMemoryOrderRepo() *InMemoryOrderRepo {  
    return &InMemoryOrderRepo{  
       Orders: make(map[int]order.Order),  
    }  
}  
  
func (repo *InMemoryOrderRepo) Add(order order.Order) {  
    defer repo.mu.Unlock()  
    repo.mu.Lock()  
    repo.Orders[order.ID] = order  
}  
  
func (repo *InMemoryOrderRepo) GetByID(id int) (order.Order, error) {  
    defer repo.mu.Unlock()  
    repo.mu.Lock()  
    value, ok := repo.Orders[id]  
    if !ok {  
       return value, order.ErrOrderNotFound  
    }  
    return value, nil  
}  
  
func (repo *InMemoryOrderRepo) Update(o order.Order) error {  
    defer repo.mu.Unlock()  
    repo.mu.Lock()  
    if _, ok := repo.Orders[o.ID]; !ok {  
       return order.ErrOrderNotFound  
    }  
    repo.Orders[o.ID] = o  
    return nil  
}  
  
func (repo *InMemoryOrderRepo) GetAll() []order.Order {  
    defer repo.mu.Unlock()  
    repo.mu.Lock()  
    orders := make([]order.Order, 0, len(repo.Orders))  
    for _, o := range repo.Orders {  
       orders = append(orders, o)  
    }  
    return orders  
}
```

# **`order_server.go`**:
```Go
package service  
  
import (  
    "fmt"  
    "gotracker/internal/order"
    "gotracker/internal/repository"
    "sync"
    )

type OrderService struct {  
    repo repository.Repository  
}  
  
func NewOrderService(repo repository.Repository) *OrderService {  
    return &OrderService{repo: repo}  
}  
  
func (service *OrderService) DeliverMany(ids []int) {  
    var wg sync.WaitGroup  
    for _, id := range ids {  
       wg.Add(1)  
       go func() {  
          defer wg.Done()  
          order, errGet := service.repo.GetByID(id)  
          if errGet != nil {  
             fmt.Printf("Ошибка: %s (ID: %v)\n", errGet, id)  
             return  
          }  
          order.MarkDelivered()  
          if errUpdate := service.repo.Update(order); errUpdate != nil {  
             fmt.Printf("Ошибка: %s (ID: %v)\n", errUpdate, id)  
             return  
          }  
          fmt.Printf("Updated order %v\n", order.ID)  
       }()  
    }  
    wg.Wait()  
}  
func (service *OrderService) PrintAllOrders() {  
    orders := service.repo.GetAll()  
    for _, v := range orders {  
       fmt.Printf("ID: %v, %v, Delivered: %v\n", v.ID, v.Customer, v.IsDelivered)  
    }  
}  
  
func (s *OrderService) GetAll() []order.Order {  
    return s.repo.GetAll()  
}  
  
func (s *OrderService) AddOrder(ord order.Order) {  
    s.repo.Add(ord)  
}  
  
func (s *OrderService) UpdateOrder(o order.Order) error {  
    return s.repo.Update(o)  
}
```
