## RESTAPI /
###### `main.go`:
```Go
package main  
  
import (  
    "fmt"  
    "net/http"
    "restapi/handlers"
    "restapi/middleware"
    "restapi/utils"
)

func main() {  
    mux := http.NewServeMux()  
    mux.HandleFunc("/slow", handlers.HandleSlowBook)  
    mux.HandleFunc("/books", func(w http.ResponseWriter, r *http.Request) {  
       switch r.Method {  
       case http.MethodGet:  
          handlers.GetBooks(w, r)  
       case http.MethodPost:  
          handlers.CreateBook(w, r)  
       default:  
          utils.WriteError(w, http.StatusMethodNotAllowed, "Метод не поддерживается")  
       }  
    })  
  
    mux.HandleFunc("/books/", func(w http.ResponseWriter, r *http.Request) {  
       switch r.Method {  
       case http.MethodGet:  
          handlers.GetBookByID(w, r)  
       case http.MethodPut:  
          handlers.UpdateBook(w, r)  
       case http.MethodDelete:  
          handlers.DeleteBook(w, r)  
       default:  
          utils.WriteError(w, http.StatusMethodNotAllowed, "Метод не поддерживается")  
       }  
    })  
    var handler http.Handler = mux  
    handler = middleware.LoggingMiddleware(handler)  
    handler = middleware.RecoveryMiddleware(handler)  
    fmt.Println("Сервер запущен на http://localhost:8080")  
  
    err := http.ListenAndServe(":8080", handler)  
    if err != nil {  
       fmt.Println("Ошибка запуска сервера:", err)  
    }  
}
```

## / handlers /
###### `books.go`:
```Go
package handlers  
  
import (  
    "encoding/json"  
    "net/http"
    "restapi/models"
    "restapi/utils"
    "strconv"
    "strings"
)

var books = []models.Book{  
    {ID: 1, Title: "Мастер и Маргарита", Author: "Булгаков", Available: true},  
    {ID: 2, Title: "Гарри Поттер", Author: "Роулинг", Available: false},  
}  
  
func GetBooks(w http.ResponseWriter, r *http.Request) {  
    w.Header().Set("Content-Type", "application/json")  
    json.NewEncoder(w).Encode(books)  
}  
  
func GetBookByID(w http.ResponseWriter, r *http.Request) {  
    idStr := strings.TrimPrefix(r.URL.Path, "/books/")  
    id, err := strconv.Atoi(idStr)  
    if err != nil {  
       utils.WriteError(w, http.StatusBadRequest, "Некорректный ID")  
       return  
    }  
  
    for _, t := range books {  
       if t.ID == id {  
          w.Header().Set("Content-Type", "application/json")  
          json.NewEncoder(w).Encode(t)  
          return  
       }  
    }  
    utils.WriteError(w, http.StatusNotFound, "Книга не найдена")  
}  
  
func CreateBook(w http.ResponseWriter, r *http.Request) {  
    var newBook models.Book  
    if err := json.NewDecoder(r.Body).Decode(&newBook); err != nil {  
       utils.WriteError(w, http.StatusBadRequest, "Невалидный JSON")  
       return  
    }  
    if err := utils.ValidateBook(newBook); err != nil {  
       utils.WriteError(w, http.StatusBadRequest, err.Error())  
       return  
    }  
    newBook.ID = len(books) + 1  
    books = append(books, newBook)  
  
    w.Header().Set("Content-Type", "application/json")  
    w.WriteHeader(http.StatusCreated)  
    json.NewEncoder(w).Encode(newBook)  
}  
  
func UpdateBook(w http.ResponseWriter, r *http.Request) {  
    idStr := strings.TrimPrefix(r.URL.Path, "/books/")  
    id, err := strconv.Atoi(idStr)  
    if err != nil {  
       utils.WriteError(w, http.StatusBadRequest, "Некорректный ID")  
       return  
    }  
  
    var updated models.Book  
    if err := json.NewDecoder(r.Body).Decode(&updated); err != nil {  
       utils.WriteError(w, http.StatusBadRequest, "Невалидный JSON")  
       return  
    }  
    if err := utils.ValidateBook(updated); err != nil {  
       utils.WriteError(w, http.StatusBadRequest, err.Error())  
       return  
    }  
  
    for i, t := range books {  
       if t.ID == id {  
          books[i].Title = updated.Title  
          books[i].Author = updated.Author  
          books[i].Available = updated.Available  
          w.Header().Set("Content-Type", "application/json")  
          json.NewEncoder(w).Encode(books[i])  
          return  
       }  
    }  
  
    utils.WriteError(w, http.StatusNotFound, "Книга не найдена")  
}  
  
func DeleteBook(w http.ResponseWriter, r *http.Request) {  
    idStr := strings.TrimPrefix(r.URL.Path, "/books/")  
    id, err := strconv.Atoi(idStr)  
    if err != nil {  
       utils.WriteError(w, http.StatusBadRequest, "Некорректный ID")  
       return  
    }  
  
    for i, t := range books {  
       if t.ID == id {  
          books = append(books[:i], books[i+1:]...)  
          w.WriteHeader(http.StatusNoContent)  
          return  
       }  
    }  
  
    utils.WriteError(w, http.StatusNotFound, "Книга не найдена")  
}
```

###### `slow.go`:
```Go
package handlers  
  
import (  
    "context"  
    "fmt"
    "net/http"
    "time"
)

func HandleSlowBook(w http.ResponseWriter, r *http.Request) {  
    ctx, cancel := context.WithTimeout(r.Context(), 3*time.Second)  
    defer cancel()  
    fmt.Println("Начали обработку запроса...")  
    select {  
    case <-time.After(5 * time.Second):  
       w.WriteHeader(http.StatusOK)  
       _, err := w.Write([]byte("Обработка выполнена"))  
       if err != nil {  
          return  
       }  
    case <-ctx.Done():  
       if ctx.Err() == context.Canceled {  
          fmt.Println("Клиент отменил операцию:", ctx.Err())  
       } else if ctx.Err() == context.DeadlineExceeded {  
          fmt.Println("Операция заняла слишком много времени:", ctx.Err())  
       }  
       return  
    }  
}
```

## / middleware /

###### `logging.go`:
```Go
package middleware  
  
import (  
    "log"  
    "net/http"
    "time"
)

func LoggingMiddleware(next http.Handler) http.Handler {  
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {  
       start := time.Now()  
       next.ServeHTTP(w, r)  
       log.Printf("%s %s - %v", r.Method, r.URL.Path, time.Since(start))  
    })  
}
```

###### `recovery.go`:
```Go
package middleware  
  
import (  
    "log"  
    "net/http"
    "restapi/utils"
)

func RecoveryMiddleware(next http.Handler) http.Handler {  
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {  
       defer func() {  
          if err := recover(); err != nil {  
             log.Printf("Паника: %v", err)  
             utils.WriteError(w, http.StatusInternalServerError, "Внутренняя ошибка сервера")  
          }  
       }()  
       next.ServeHTTP(w, r)  
    })  
}
```

```Go
package utils  
  
import (  
    "encoding/json"  
    "errors"
    "net/http"
    "restapi/models"
)

func ValidateBook(b models.Book) error {  
    if b.Title == "" {  
       return errors.New("Поле 'Title' не может быть пустым.")  
    }  
    if len(b.Title) > 100 {  
       return errors.New("Название книги не должно превышать 100 символов")  
    }  
    if b.Author == "" {  
       return errors.New("Поле 'Author' не может быть пустым.")  
    }  
    return nil  
}  
  
type ErrorResponse struct {  
    Error string `json:"error"`  
}  
  
func WriteError(w http.ResponseWriter, code int, msg string) {  
    w.Header().Set("Content-Type", "application/json")  
    w.WriteHeader(code)  
    json.NewEncoder(w).Encode(ErrorResponse{Error: msg})  
}
```
